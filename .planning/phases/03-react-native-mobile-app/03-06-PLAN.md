## Threat Model

| Boundary | Description |
|----------|-------------|
| Mobile SSE client → /api/chat-stream/:tradeId | SSE stream must verify trade participant before streaming |
| Mobile → trpc.trades.agreeToTrade | Must verify caller is a trade participant server-side |

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-06-01 | Spoofing | SSE endpoint authentication | mitigate | useChatStream sends `Authorization: Bearer <token>` header; the existing SSE endpoint calls `requireAuth(request)` which Better Auth expo() plugin processes for Bearer tokens |
| T-03-06-02 | Information Disclosure | Trade message visibility | mitigate | trpc.trades.getById already checks `initiatorId === userId || counterpartyId === userId` (Plan 02); SSE endpoint has the same check — only stream messages for trades the user is party to |
| T-03-06-03 | Tampering | trpc.trades.agreeToTrade double-tap | mitigate | Server checks if user already agreed (`initiatorAgreed` / `counterpartyAgreed`). Repeat mutations are idempotent — no harm in double-tapping the Agree button |
| T-03-06-04 | Denial of Service | SSE reconnection storm | mitigate | @microsoft/fetch-event-source has exponential backoff built in. useChatStream AbortController prevents reconnection after component unmount |

    CRITICAL: Use `@microsoft/fetch-event-source` NOT `@azure/fetch-event-source` (which 404s).
    ```typescript
    import { useEffect, useRef, useCallback } from "react";
    import { fetchEventSource } from "@microsoft/fetch-event-source";
    import { authClient } from "~/lib/auth";

    interface ChatStreamMessage {
      type: "message" | "status";
      data: unknown;
    }

    export function useChatStream(
      tradeId: number | null,
      onMessage: (msg: ChatStreamMessage) => void,
    ) {
      const abortRef = useRef<AbortController | null>(null);
      const onMessageRef = useRef(onMessage);
      onMessageRef.current = onMessage;

      useEffect(() => {
        if (!tradeId) return;

        const ctrl = new AbortController();
        abortRef.current = ctrl;

        async function connect() {
          const session = await authClient.getSession();
          const token = (session as { token?: string } | null)?.token;

          await fetchEventSource(
            `${process.env.EXPO_PUBLIC_API_URL}/api/chat-stream/${tradeId}`,
            {
              method: "GET",
              headers: {
                Authorization: token ? `Bearer ${token}` : "",
                Accept: "text/event-stream",
                "Cache-Control": "no-cache",
              },
              signal: ctrl.signal,
              onmessage(ev) {
                if (!ev.data) return;
                try {
                  const parsed = JSON.parse(ev.data) as ChatStreamMessage;
                  onMessageRef.current(parsed);
                } catch {
                  // Non-JSON event (ping keepalive etc.) — ignore
                }
              },
              onerror(err) {
                // fetchEventSource auto-retries with backoff; this is a warning only
                console.warn("[SSE] chat stream error, retrying...", err);
                // Do NOT throw here — throwing stops retry
              },
              onclose() {
                console.log("[SSE] chat stream closed for trade", tradeId);
              },
            }
          );
        }

        connect().catch((err) => {
          if (err?.name !== "AbortError") {
            console.error("[SSE] fatal error:", err);
          }
        });

        return () => {
          ctrl.abort();
        };
      }, [tradeId]);
    }
    ```

    ```typescript
    import {
      View, Text, TextInput, TouchableOpacity,
      KeyboardAvoidingView, Platform, ActivityIndicator, Alert
    } from "react-native";
    import { SafeAreaView } from "react-native-safe-area-context";
    import { useLocalSearchParams, router } from "expo-router";
    import { ArrowLeft, Send } from "lucide-react-native";
    import { trpc } from "~/lib/trpc";
    import { authClient } from "~/lib/auth";
    import { ChatWindow } from "~/components/ui/ChatWindow";
    import { HandshakeFlow } from "~/components/ui/HandshakeFlow";
    import { useChatStream } from "~/hooks/useChatStream";
    import type { ChatMessage } from "@nozar/shared/types";

    export default function PingDetailScreen() {
      const { id } = useLocalSearchParams<{ id: string }>();
      const { data: session } = authClient.useSession();
      const [messageText, setMessageText] = useState("");
      const [streamMessages, setStreamMessages] = useState<ChatMessage[]>([]);

      const utils = trpc.useUtils();
      const { data, isLoading } = trpc.trades.getById.useQuery(
        { id: Number(id) },
        { enabled: !!id }
      );

      const addMessageMutation = trpc.trades.addMessage.useMutation({
        onSuccess: () => {
          setMessageText("");
          utils.trades.getById.invalidate({ id: Number(id) });
        },
      });

      const agreeMutation = trpc.trades.agreeToTrade.useMutation({
        onSuccess: () => {
          utils.trades.getById.invalidate({ id: Number(id) });
        },
        onError: (err) => Alert.alert("Could not agree", err.message),
      });

      const updateStatusMutation = trpc.trades.updateStatus.useMutation({
        onSuccess: () => {
          utils.trades.getById.invalidate({ id: Number(id) });
        },
      });

      // SSE hook for real-time messages
      const handleStreamMessage = useCallback(
        (msg: { type: string; data: unknown }) => {
          if (msg.type === "message" && msg.data) {
            const chatMsg = msg.data as ChatMessage;
            setStreamMessages((prev) => {
              // Deduplicate by id
              if (prev.some((m) => m.id === chatMsg.id)) return prev;
              return [...prev, chatMsg];
            });
          } else if (msg.type === "status") {
            // Trade status update — refetch
            utils.trades.getById.invalidate({ id: Number(id) });
          }
        },
        [id, utils]
      );

      useChatStream(id ? Number(id) : null, handleStreamMessage);

      if (isLoading) {
        return (
          <SafeAreaView className="flex-1 bg-[#030912] items-center justify-center">
            <ActivityIndicator color="#10b981" />
          </SafeAreaView>
        );
      }

      if (!data) {
        return (
          <SafeAreaView className="flex-1 bg-[#030912] items-center justify-center">
            <Text className="text-slate-400">Trade not found</Text>
          </SafeAreaView>
        );
      }

      const { trade, messages: trpcMessages } = data;
      const currentUserId = session?.user?.id ?? "";
      const isInitiator = trade.initiatorId === currentUserId;

      // Merge tRPC messages with SSE stream messages (deduplicated)
      const allMessages: ChatMessage[] = [
        ...(trpcMessages ?? []).map((m) => ({
          id: m.id,
          senderId: m.senderId,
          senderName: m.senderName ?? "Unknown",
          text: m.text,
          createdAt: m.createdAt.toString(),
        })),
        ...streamMessages.filter(
          (sm) => !(trpcMessages ?? []).some((m) => m.id === sm.id)
        ),
      ].sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());

      function handleSend() {
        if (!messageText.trim() || !id) return;
        addMessageMutation.mutate({ tradeId: Number(id), text: messageText.trim() });
      }

      function handleAgree() {
        if (!id) return;
        agreeMutation.mutate({ tradeId: Number(id) });
      }

      function handleShareContact() {
        if (!id) return;
        updateStatusMutation.mutate({ tradeId: Number(id), status: "contact_shared" });
      }

      function handleDecline() {
        Alert.alert(
          "Decline trade?",
          "This will cancel the trade.",
          [
            { text: "Keep trade", style: "cancel" },
            {
              text: "Yes, decline",
              style: "destructive",
              onPress: () => {
                if (id) updateStatusMutation.mutate({ tradeId: Number(id), status: "cancelled" });
              },
            },
    ```
